rules_version = '2';
// Tonicão Team v0.29 — multiacademia, separação Plataforma/Academia e consentimento.
service cloud.firestore {
  match /databases/{database}/documents {
    function signed() { return request.auth != null; }
    function platformPath() { return /databases/$(database)/documents/platformAdmins/$(request.auth.uid); }
    function isPlatformAdmin() { return signed() && exists(platformPath()); }
    function academyPath(a) { return /databases/$(database)/documents/academies/$(a); }
    function memberPath(a) { return /databases/$(database)/documents/academies/$(a)/members/$(request.auth.uid); }
    function mem(a) { return get(memberPath(a)).data; }
    function isMember(a) { return signed() && exists(memberPath(a)) && mem(a).active == true; }
    function isProf(a) { return isMember(a) && mem(a).role == 'professor'; }
    function isAdmin(a) { return isMember(a) && mem(a).role == 'admin'; }
    function isStaff(a) { return isProf(a) || isAdmin(a); }
    function isAluno(a) { return isMember(a) && mem(a).role == 'aluno'; }
    function sid(a) { return mem(a).studentId; }
    function notBlocked(a) { return isPlatformAdmin() || isAdmin(a) || get(academyPath(a)).data.get('accessStatus', 'active') != 'blocked'; }
    function stamped() { return request.resource.data._srv == request.time && request.resource.data._by == request.auth.uid; }
    function shortStr(v, n) { return v is string && v.size() <= n; }

    match /platformAdmins/{uid} {
      allow get: if signed() && request.auth.uid == uid;
      allow list, create, update, delete: if false;
    }

    match /academies/{a} {
      allow get: if isPlatformAdmin() || isMember(a) || (signed() && resource.data.ownerUid == request.auth.uid);
      allow list: if isPlatformAdmin();
      allow create: if signed()
                    && request.resource.data.ownerUid == request.auth.uid
                    && request.resource.data.accessStatus == 'active';
      allow update: if isPlatformAdmin()
        || (isAdmin(a)
          && request.resource.data.ownerUid == resource.data.ownerUid
          && !request.resource.data.diff(resource.data).affectedKeys().hasAny(
            ['accessStatus','accessReason','subscriptionStatus','subscriptionUntil']
          ));
      allow delete: if false;

      match /members/{uid} {
        allow get: if signed() && (request.auth.uid == uid || isStaff(a));
        allow list: if isStaff(a);
        allow create: if signed() && (
          (uid == request.auth.uid && get(academyPath(a)).data.ownerUid == request.auth.uid
            && request.resource.data.role == 'admin' && request.resource.data.active == true)
          || (uid == request.auth.uid && request.resource.data.role == 'aluno' && request.resource.data.active == false
            && request.resource.data.studentId == ''
            && request.resource.data.keys().hasOnly(['uid','name','email','picture','provider','role','studentId','active','createdAt','_srv','_by']))
          || isAdmin(a)
          || (isProf(a) && request.resource.data.role == 'aluno')
        );
        allow update: if isAdmin(a)
          || (isProf(a) && resource.data.role == 'aluno' && request.resource.data.role == 'aluno');
        allow delete: if isAdmin(a) && uid != request.auth.uid;
      }

      match /registrationInvites/{token} {
        allow get: if true;
        allow list: if isStaff(a);
        allow create: if (
          isProf(a)
          && request.resource.data.status == 'invited'
          && request.resource.data.expiresAtMs is int
          && request.resource.data.expiresAtMs > request.time.toMillis()
          && request.resource.data.expiresAtMs <= request.time.toMillis() + 604800000
        ) || (
          request.resource.data.status == 'pending'
          && request.resource.data.source == 'self'
          && request.resource.data.keys().hasOnly([
            'name','phone','birth','graduationTrack','guardianName','guardianPhone','note',
            'privacyConsentAt','guardianConsentAt','status','source','inviteToken','createdAt','_srv','_by'
          ])
          && shortStr(request.resource.data.name,120)
          && request.resource.data.name.size() > 1
          && shortStr(request.resource.data.phone,40)
          && shortStr(request.resource.data.note,300)
          && request.resource.data.privacyConsentAt is string
          && request.resource.data.privacyConsentAt.size() > 0
          && (request.resource.data.graduationTrack != 'kids'
              || (shortStr(request.resource.data.guardianName,120)
                  && request.resource.data.guardianName.size() > 1
                  && shortStr(request.resource.data.guardianPhone,40)
                  && request.resource.data.guardianPhone.size() > 3
                  && request.resource.data.guardianConsentAt is string
                  && request.resource.data.guardianConsentAt.size() > 0))
        );
        allow update: if (
          isProf(a)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
            ['status','studentId','decidedAt','_srv','_by']
          )
          && request.resource.data.status in ['approved','rejected']
        ) || (
          resource.data.status == 'invited'
          && request.resource.data.status == 'awaiting_approval'
          && resource.data.get('expiresAtMs',0) >= request.time.toMillis()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
            'name','phone','email','birth','graduationTrack','belt','stripes',
            'guardianName','guardianPhone','emergencyName','emergencyPhone','note',
            'privacyConsentAt','guardianConsentAt','status','completedAt','_srv','_by'
          ])
          && shortStr(request.resource.data.name,120)
          && shortStr(request.resource.data.note,300)
          && request.resource.data.privacyConsentAt is string
          && request.resource.data.privacyConsentAt.size() > 0
          && (request.resource.data.graduationTrack != 'kids'
              || (request.resource.data.guardianName.size() > 1
                  && request.resource.data.guardianPhone.size() > 3
                  && request.resource.data.guardianConsentAt.size() > 0))
        );
        allow delete: if isProf(a);
      }

      match /sessionCodes/{id} {
        allow read: if isStaff(a) && notBlocked(a);
        allow write: if isProf(a) && notBlocked(a) && stamped();
      }
      match /gradingPrivate/{id} {
        allow read: if isStaff(a) && notBlocked(a);
        allow write: if isProf(a) && notBlocked(a) && stamped();
      }

      match /attendance/{id} {
        allow read: if notBlocked(a) && (isStaff(a) || (isAluno(a) && resource.data.studentId == sid(a)));
        allow write: if isProf(a) && notBlocked(a) && stamped();
        allow create: if isAluno(a) && notBlocked(a) && stamped()
          && request.resource.data.keys().hasOnly(['id','studentId','sessionId','date','time','source','status','checkinCode','_srv','_by'])
          && request.resource.data.studentId == sid(a)
          && id == request.resource.data.sessionId + '_' + sid(a)
          && request.resource.data.status == 'pending'
          && request.resource.data.source == 'qr-codigo'
          && exists(/databases/$(database)/documents/academies/$(a)/sessionCodes/$(request.resource.data.sessionId))
          && request.resource.data.checkinCode.size() == 6
          && request.resource.data.checkinCode in [
               get(/databases/$(database)/documents/academies/$(a)/sessionCodes/$(request.resource.data.sessionId)).data.code,
               get(/databases/$(database)/documents/academies/$(a)/sessionCodes/$(request.resource.data.sessionId)).data.get('prevCode', '-'),
               get(/databases/$(database)/documents/academies/$(a)/sessionCodes/$(request.resource.data.sessionId)).data.get('nextCode', '-')]
          && request.time.toMillis() >= get(/databases/$(database)/documents/academies/$(a)/sessionCodes/$(request.resource.data.sessionId)).data.startMs
          && request.time.toMillis() <= get(/databases/$(database)/documents/academies/$(a)/sessionCodes/$(request.resource.data.sessionId)).data.endMs + 21600000
          && !(get(/databases/$(database)/documents/academies/$(a)/students/$(sid(a))).data.get('payment','liberado') in ['pendente','verificar','bloqueado']);
      }

      match /classBookings/{id} {
        allow read: if notBlocked(a) && (isStaff(a) || (isAluno(a) && resource.data.studentId == sid(a)));
        allow write: if isProf(a) && notBlocked(a) && stamped();
        allow create: if isAluno(a) && notBlocked(a) && stamped()
          && request.resource.data.keys().hasOnly(['id','scheduleId','date','studentId','status','createdAt','_srv','_by'])
          && request.resource.data.studentId == sid(a)
          && request.resource.data.status == 'pending'
          && request.resource.data.date is string && request.resource.data.date.size() == 10
          && id == request.resource.data.scheduleId + '_' + request.resource.data.date + '_' + sid(a)
          && exists(/databases/$(database)/documents/academies/$(a)/classSchedule/$(request.resource.data.scheduleId));
        allow update: if isAluno(a) && notBlocked(a) && stamped()
          && resource.data.studentId == sid(a) && resource.data.status == 'pending'
          && request.resource.data.studentId == sid(a) && request.resource.data.status == 'pending'
          && request.resource.data.scheduleId == resource.data.scheduleId && request.resource.data.date == resource.data.date
          && request.resource.data.keys().hasOnly(['id','scheduleId','date','studentId','status','createdAt','_deleted','_srv','_by']);
      }

      match /gradingAssignments/{id} {
        allow read: if notBlocked(a) && (isStaff(a) || (isAluno(a) && resource.data.studentId == sid(a)));
        allow write: if isProf(a) && notBlocked(a) && stamped();
        allow update: if isAluno(a) && notBlocked(a) && stamped() && resource.data.studentId == sid(a)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['progress','_srv','_by'])
          && request.resource.data.progress is map;
      }

      match /students/{id} {
        allow get: if notBlocked(a) && (isStaff(a) || (isAluno(a) && id == sid(a)));
        allow list: if isStaff(a) && notBlocked(a);
        allow write: if isProf(a) && notBlocked(a) && stamped();
      }
      match /studentsPublic/{id} {
        allow read: if isMember(a) && notBlocked(a);
        allow write: if isProf(a) && notBlocked(a) && stamped();
      }

      match /{store}/{id} {
        allow read: if notBlocked(a) && store in ['gradingHistory','gradingReminders','pointsLedger','paymentStatus']
          && (isStaff(a) || (isAluno(a) && resource.data.studentId == sid(a)));
        allow write: if isProf(a) && notBlocked(a) && stamped()
          && store in ['gradingHistory','gradingReminders','pointsLedger','paymentStatus'];
      }

      match /{store}/{id} {
        allow read: if isMember(a) && notBlocked(a)
          && store in ['techniques','gradingPlans','events','academyContent','academyTimeline','academyGallery','academyRules',
                       'scoreRules','referenceMaterials','timerPresets','sequenceRules','classSessions','classSchedule'];
        allow write: if notBlocked(a) && stamped() && (
          (isProf(a) && store in ['techniques','gradingPlans','events','academyContent','academyTimeline','academyGallery','academyRules',
                                  'scoreRules','referenceMaterials','timerPresets','sequenceRules','classSessions','classSchedule'])
          || (isAdmin(a) && store in ['academyContent','academyTimeline','academyGallery','referenceMaterials'])
        );
      }

      match /notifications/{id} {
        allow read: if notBlocked(a) && (isStaff(a) || (isAluno(a)
          && (resource.data.targetStudentId == sid(a) || resource.data.targetRole == 'aluno')));
        allow write: if isStaff(a) && notBlocked(a) && stamped();
      }
    }
  }
}
